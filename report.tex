\addfrontmattertotoc{ABSTRACT}
\noindent
Frequency Hopping Spread Spectrum (FHSS) is a dominant method for securing military and commercial communications against jamming and interception. However, classical pseudo-random number (PN) generators used in FHSS, such as Gold Codes, strictly follow linear recurrence relations characterized by their linear complexity. This project presents a comprehensive approach to FHSS analysis, beginning with the development of a modular MATLAB simulation framework (Phase 1) to generate high-fidelity synthetic data and validate system performance under noise. Building on this, we investigate the vulnerability of these generators to Deep Learning attacks using a hybrid CNN-Transformer architecture.
Our results demonstrate a fundamental dichotomy in physical-layer security. We show that simple M-Sequence generators are structurally brittle: once the observation window exceeds the theoretical observability threshold, our model "solves" the generator polynomial, achieving \textbf{99.9\% prediction accuracy} even on unseen communication seeds. In contrast, Gold Codes exhibited significant robustness, stalling at 25\% accuracy (a partial break) due to their disjoint cycle topology. These findings validate that while Deep Learning can act as an automated cryptanalysis tool for legacy systems, cryptographically designed sequences remain robust against black-box AI attacks.

\vspace{2cm}
\noindent Keywords: FHSS, Deep Learning, CNN-Transformer, Cryptanalysis, Sequence Prediction

% CHAPTER 1: INTRODUCTION
\chapter{INTRODUCTION}
\section{Background}
Frequency-Hopping Spread Spectrum (FHSS) is a vital technique in modern wireless communications, offering resilience against jamming and interference \cite{proakis}. By rapidly switching the carrier frequency according to a pseudo-random sequence, the transmitter makes it difficult for an adversary to jam the signal or intercept the message. In crowded spectrum environments, the ability to detect and characterize these signals is crucial for spectrum surveillance \cite{goodfellow}.

The security of an FHSS system relies entirely on the secrecy of the hopping sequence. These sequences are typically generated by Linear Feedback Shift Registers (LFSRs) combined to form Gold Codes or m-sequences \cite{golomb}. While statistically random to a casual observer, these sequences are deterministically generated by simple linear equations over a Galois Field GF(2).

\section{Problem Statement}
If an adversary can predict the next frequency hop in real-time, they can launch a "Follower Jammer" attack, concentrating their energy on the specific channel the victim is about to use, rendering the FHSS protection useless. 
Traditional cryptanalysis methods (like the Berlekamp-Massey algorithm \cite{massey}) differ fundamentally from Deep Learning. Algebraic attacks are precise but brittle—requiring error-free demodulation of the bitstream. In contrast, Deep Learning models (like Transformers \cite{attention}) offer a "soft" attack surface, potentially learning to predict hops directly from noisy spectrograms without explicit demodulation.

\section{Project Objectives}
The primary objective of this project is to investigate the vulnerability of classical FHSS generators to modern Deep Learning attacks. This was achieved in two phases:
\begin{enumerate}
    \item \textbf{Phase 1 (Simulation):} Establish a modular FHSS simulation environment in MATLAB to generate high-fidelity synthetic data containing both M-Sequences and Gold Codes under varying SNR conditions.
    \item \textbf{Phase 2 (Cryptanalysis):} Design a Hybrid \textbf{CNN-Transformer} architecture to attack these sequences. We aim to determine if the model can learn the underlying generator polynomial purely from observation, and to quantify the "Unlearnability" of Gold Codes compared to simple M-Sequences.
\end{enumerate}

\chapter{ANALYSIS}
\section{FHSS Signal Model}
The FHSS signal is modeled as a sequence of modulated pulses, each transmitted on a different carrier frequency. The transmitted signal $s(t)$ can be expressed as:
\begin{equation}
    s(t)=\sum_{k=-\infty}^{\infty}p(t-kT_{h})\cos(2\pi f_{k}(t-kT_{h})+\theta_{k}(t)+\phi_{k})
\end{equation}
where $T_{h}$ is the hop duration (dwell time), $p(t)$ is a rectangular pulse function, and $f_{k}$ is the carrier frequency selected from a predefined hopset. In our binary FSK implementation, the instantaneous frequency is $f_{k}\pm\Delta f/2$.

\section{Galois Field Arithmetic (GF(2))}
All standard cryptographic sequence generators (LFSRs, Gold Codes) operate over the Galois Field of two elements, denoted as $\text{GF}(2)$ or $\mathbb{F}_2$. This field consists of the set $\{0, 1\}$ equipped with two operations:
\begin{itemize}
    \item \textbf{Addition (+):} Corresponds to the logical \textbf{XOR} operation ($1+1=0$).
    \item \textbf{Multiplication ($\cdot$):} Corresponds to the logical \textbf{AND} operation.
\end{itemize}
Crucially, subtraction in $\text{GF}(2)$ is identical to addition. This algebraic structure simplifies hardware implementation but also makes the generation process strictly linear, which is the foundational weakness exploited in this project.

\section{The Mathematical Vulnerability of FHSS}
The core vulnerability of most FHSS systems is the linearity of their generators. A Gold Code Generator consists of two LFSRs combined via XOR.
Let the state of an LFSR at time $t$ be $S_t$. The next state is determined by a linear recurrence:
\begin{equation}
    S_{t+1} = \sum_{i=1}^{L} c_i S_{t-i} \pmod 2
\end{equation}
Because the operation is purely linear (XOR is equivalent to addition in GF(2)), the "Linear Complexity" of the system is finite. For a Gold Code generated by two degree-$n$ polynomials, the linear complexity is exactly $2n$. This implies that observing $2n$ consecutive bits allows one to mathematically solve for the entire future sequence.

\section{The Berlekamp-Massey Algorithm}
The Berlekamp-Massey (BM) algorithm is the standard cryptanalytic tool for breaking LFSR-based streams. Given a binary sequence $s_0, s_1, \dots, s_{N-1}$, the algorithm constructs the shortest LFSR that can generate this sequence.
\begin{itemize}
    \item \textbf{Input:} A stream of bits.
    \item \textbf{Output:} The connection polynomial $C(D)$ and the linear complexity $L$.
    \item \textbf{Theorem:} If the sequence is generated by an LFSR of length $L$, the BM algorithm uniquely determines the polynomial after processing exactly $2L$ bits.
\end{itemize}
While mathematically perfect ($O(N^2)$ complexity), BM has a fatal flaw in Electronic Warfare scenarios: \textbf{it is brittle}. A single bit error in the input stream causes the calculated linear complexity to explode, rendering the result useless. This necessitates a "soft" approach (like Deep Learning) that can tolerate observation noise (AWGN) while still approximating the underlying rule.

\section{Why Gold Codes Break Deep Learning (But M-Sequences Don't)}
Our experiments revealed a critical distinction: M-Sequences ($L=10$) were solved to 99.9\% accuracy, while Gold Codes ($L=20$) stalled at 25\%. This refutes the common assumption that Neural Networks simply "cannot learn XOR" or "cannot handle Finite Fields." They \textit{can}, as evidenced by the M-Sequence result. The failure on Gold Codes stems from two specific properties:

\subsection{Topological Disconnectedness}
M-Sequences define a single connected state-space cycle (the "Universe"). Training on any segment provides information about the global structure.
Gold Codes, however, define a "Multiverse" of disjoint cycles. The mapping $f: \text{History} \to \text{Next State}$ is consistent across cycles, but the data distribution is fragmented. The optimizer must learn a function that generalizes across these disconnected manifolds. Our results suggest that standard Gradient Descent struggles to interpolate between these disjoint cycles, effectively learning "local" statistics (leading to 25\% accuracy) rather than the global algebraic rule.

\subsection{Linear Complexity Scaling}
The "Linear Complexity" $L$ defines the dimension of the state space.
\begin{itemize}
    \item \textbf{M-Sequence:} $L=10$. Search space $\approx 2^{10} = 1024$ states.
    \item \textbf{Gold Code:} $L=20$. Search space $\approx 2^{20} \approx 1,000,000$ states.
\end{itemize}
While 42 bits of history theoretically resolve $2^{20}$ states, the neural network does not perform Gaussian Elimination. It performs smooth optimization on a non-smooth (shattered) landscape. The transition from $10^{3}$ inputs to $10^{6}$ inputs likely pushes the problem beyond the "Grokking" threshold of the Transformer within the limited training epochs, turning a tractable problem into an intractable one for the optimizer. This validates the use of high-complexity sequences for security: they force the attacker to perform exponential work ($O(2^L)$) or brittle algebraic inversion ($O(L^2)$), characterizing the "Cryptographic Strength" of the code against AI approximation.


\section{Time-Frequency Analysis: STFT}
To analyze non-stationary FHSS signals, we employ the Short-Time Fourier Transform (STFT):
\begin{equation}
    STFT_{x}(t,f)=\int_{-\infty}^{\infty}x(\tau)w(\tau-t)^{*}e^{-j2\pi f\tau}d\tau
\end{equation}
The spectrogram, $|STFT_{x}(t,f)|^{2}$, provides a visual representation of energy distribution. However, it is limited by the Heisenberg-Gabor uncertainty principle. A wide window provides good frequency resolution but poor time resolution (smearing transitions), while a narrow window offers the inverse.

\chapter{DEEP LEARNING METHODOLOGY}
\section{Dataset Generation}
The dataset consists of FHSS signals generated using Gold Codes of degree 10.
\begin{itemize}
    \item \textbf{Signal Parameters:} $f_s = 10$ MHz, Hop Rate $= 100$ hops/sec.
    \item \textbf{Modulation:} Binary FSK with a frequency separation equal to the symbol rate.
    \item \textbf{Noise:} AWGN added at SNR levels from -10 dB to 10 dB.
    \item \textbf{STFT Slicing (The "Vision" Input):} The raw time-domain signal is converted to a spectrogram using the STFT (Window=128, Overlap=96). We explicitly slice the spectrogram into chunks corresponding to a single hop duration. This results in an input dimension of $1024 \times 3$ (1024 frequency bins, 3 time columns). The CNN sees only this localized snapshot, forcing it to classify the frequency based on a static 3-column image rather than temporal evolution.
\end{itemize}

\section{System Architecture}
We implemented a hybrid architecture composed of two distinct modules: a Vision Module for signal detection and a Sequence Module for cryptographic analysis.

\subsection{Vision Module (CNN)}
The Convolutional Neural Network (CNN) is responsible for extracting the current frequency from the spectrogram.
\begin{itemize}
    \item \textbf{Input:} Single STFT frame (1024 bins $\times$ 3 time steps).
    \item \textbf{Layers:} 4-Layer Deep Convolutional Network (Conv2d $\to$ InstanceNorm $\to$ ReLU $\to$ MaxPool).
    \item \textbf{Symbolic Interface:} Unlike standard feature extraction, we trained the CNN to output a discrete \textbf{Frequency Index} (Integers 0-7). This "Hard Decision" step was introduced to isolate the noise problem from the logic problem.
\end{itemize}

\subsection{Sequence Modeler (Transformer)}
We successfully implemented a Transformer-based sequence modeler designed to explicitly solve the "Tap finding" problem.
\begin{itemize}
    \item \textbf{Why Transformers?:} Unlike LSTMs, which must compress history into a fixed hidden state $h_t$, Transformers process the entire history window simultaneously. The Self-Attention mechanism computes a weighted sum of all past inputs.
    $$ Attention(Q, K, V) = softmax(\frac{QK^T}{\sqrt{d_k}})V $$
    This allows the model to learn to "attend" specifically to the relevant LFSR taps (e.g., $t-3$ and $t-10$) regardless of how far back they are, solving the vanishing gradient problem for XOR operations.
    \item \textbf{Architecture:}
    \begin{itemize}
        \item \textbf{Embeddings:} Discrete frequency indices mapped to vectors ($d=64$).
        \item \textbf{Positional Encoding:} Essential for giving the model a sense of relative time (e.g., "10 steps ago").
        \item \textbf{Transformer Encoder:} 2 Layers, 4 Heads, 128-dim Feedforward. 4 Heads were chosen to ensure independent attention mechanisms could track multiple feedback taps simultaneously.
    \end{itemize}
\end{itemize}

\section{Experimental Methodology & Results}
We conducted a series of experiments to isolate the capabilities of the model.

\subsection{Experiment 1-4: Gold Code Challenge}
As detailed previously, all attempts to train on the full Gold Code dataset (LSTM, Reduced Complexity, Static Embeddings, Transformers) resulted in non-convergence (Random Guessing, $\sim12.5\%$ accuracy). This confirmed the cryptanalytic strength of the Gold Code's disjoint cycle structure.

\subsection{Experiment 5: M-Sequence Control (Validation)}
To prove that our Transformer architecture \textit{is} capable of learning LFSR logic (and that Experiment 1-4 failed due to the specific difficulty of Gold Codes), we generated a control dataset using a simple M-Sequence ($P(x) = x^{10} + x^3 + 1$).
\begin{itemize}
    \item \textbf{Hypothesis:} Since M-Sequences have a single cycle and lower linear complexity (10 vs 20), the model should be able to generalize from seeing 70\% of the seeds (initial states) to the unseen 30\%.
    \item \textbf{Setup:} Identical Transformer architecture from Exp. 4.
    \item \textbf{Result:} The model successfully broke the random guess barrier, achieving \textbf{>99\% accuracy} on the validation set.
    \item \textbf{Conclusion:} This conclusively proves that Neural Networks \textit{can} approximate LFSR logic over GF(2) provided the state space is traversable (single cycle).
\end{itemize}

\subsection{Sensitivity to Lookback Window (Verification of Observability)}
To rigorously verify that the model was learning the underlying algebraic rule rather than statistical correlations, we varied the Lookback Window size ($W$) for the M-sequence experiment.
Math dictates that for a degree-10 LFSR ($L=10$), we need at least 10 bits of history to determine the state. Since each hop provides 3 bits ($\log_2 8$), the theoretical threshold is $\lceil 10/3 \rceil = 4$ hops.

\chapter{CONCLUSION}
\section{Results}
Our findings present a bifurcated reality for physical-layer security:
\begin{enumerate}
    \item \textbf{Vulnerability of M-Sequences:} We demonstrated that simple linear sequences (Degree 10 LFSRs) are structurally brittle against deep learning. Once the observation window exceeded the theoretical observability threshold ($W \ge 4$), the model transitioned instantly from random guessing to \textbf{99.9\% accuracy}, effectively "solving" the finite field algebra.
    \item \textbf{Robustness of Gold Codes:} In stark contrast, Gold Codes resisted a full break. Despite having sufficient information ($W=14$, 42 bits) to theoretically solve the 20-bit state, the model plateaued at \textbf{25\% accuracy}. This represents a "Partial Break" (predicting 1 bit of entropy) but confirms that the topological complexity of Gold Codes acting as a "One-Way Function" is robust against standard gradient-based learning.
\end{enumerate}

\section{Implications}
This split result is significant. It demonstrates that:
\begin{enumerate}
    \item \textbf{Limit of Gradient Descent:} While Neural Networks are universal approximators, they effectively fail to traverse the shattered loss landscape of high-order parity functions (like the XOR sum of two sequences) without specialized architectures.
    \item \textbf{Validation of Gold Codes:} The classical design of FHSS generators remains highly robust against "naive" Deep Learning attacks, verifying their continued utility in secure communications.
    \item \textbf{Cryptanalytic Potential:} Deep Learning can act as an automated cryptanalysis tool for simple systems (M-sequences), but complexity scaling protects stronger codes.
\end{enumerate}

\section{Realistic Constraints}
\subsection{Hardware Bandwidth}
Real-time detection of fast frequency-hopping signals requires processing wideband spectrum data. Capturing the entire hopping range of modern systems (e.g., Bluetooth's 80 MHz band) requires high-sample-rate Software-Defined Radios (SDRs). Low-cost SDRs have limited instantaneous bandwidth (approx. 2 MHz), constraining real-world testing to narrowband sub-sections.

\subsection{Social, Environmental and Economic Impact}
\begin{itemize}
    \item \textbf{Social:} Reliable detection enhances security by identifying unauthorized transmissions (e.g., illicit drone controllers).
    \item \textbf{Environmental:} Improved detection tools contribute to better spectrum management policies, reducing spectral congestion.
    \item \textbf{Economic:} The project utilizes software simulation, minimizing direct material costs. The developed algorithms have potential economic value for spectrum monitoring authorities.
\end{itemize}

\section{Standards}
The design adheres to standard digital signal processing practices and is informed by relevant international standards:
\begin{itemize}
    \item \textbf{IEEE 802.15.1 (Bluetooth):} Defines a widely used commercial FHSS standard operating in the 2.4 GHz ISM band with 1600 hops/sec.
    \item \textbf{Link 16 (MIL-STD-6016):} A military standard employing fast frequency-hopping (over 77,000 hops/sec) for high anti-jam resistance.
\end{itemize}

\chapter{Gold Codes vs. M-Sequences} \label{app:A}
\section{Disjoint Cycles}
A Maximal Length Sequence (m-sequence) has a single cycle of length $2^n - 1$. If you train on any part of it, you effectively see the whole "universe" of that sequence.
Gold Codes, however, are formed by adding two m-sequences. They form a family of many distinct, disjoint cycles.

\section{The "Universe" Problem}
The fundamental difference between our two experiments lies in the state-space topology.

\subsection{M-Sequences: Single Connected Component}
A Maximal Length Sequence (m-sequence) of degree $n$ has a period of $P = 2^n - 1$. Crucially, it contains \textbf{every possible non-zero state vector} in a single cycle.
\begin{equation}
    \mathcal{S}_{mseq} = \{ \text{Cycle}_1 \}
\end{equation}
When we train on a subset of seeds (70\%), we are effectively training on 70\% of the \textit{only} cycle that exists. The "Test Set" (30\%) consists of holes in that same cycle. The model only needs to learn local transitions ($S_t \to S_{t+1}$), which are globally consistent. This is why we achieved 99.9\% accuracy.

\subsection{Gold Codes: Disjoint Connected Components}
Gold Codes are generated by $G(x) = M_1(x) \oplus M_2(x)$. This addition creates a family of sequences with period $2^n - 1$, but unlike m-sequences, the state space is partitioned into many \textbf{disjoint cycles}.
\begin{equation}
    \mathcal{S}_{gold} = \{ \text{Cycle}_A, \text{Cycle}_B, \text{Cycle}_C, \dots \}
\end{equation}
If the training set consists of seeds from $\{ \text{Cycle}_A, \text{Cycle}_B \}$, and the test set consists of seeds from $\{ \text{Cycle}_C \}$, the model faces a much harder generalization task. It cannot simply memorize the trajectory; it must learn the abstract algebraic rule $G(x)$ that governs \textit{all} cycles.
Our result (25\% accuracy) suggests that Gradient Descent struggles to cross the "gap" between these disjoint cycles, managing to learn only the most dominant linear features (1 bit) while failing to capture the full 20-bit state dynamics required for perfect prediction.

\chapter{Theoretical Fundamentals of Sequence Observability} \label{app:B}
The mathematical framework presented in this appendix relies on the foundational theory of Linear Feedback Shift Registers established by Golomb \cite{golomb}. The concepts of observability and polynomial synthesis are derived from the seminal work of Massey \cite{massey} and standard Linear System Theory over Finite Fields.

\section{Notation and Model}

All algebra is over $\F_2=\GF(2)$.

\subsection{Single LFSR (m-sequence)}
Let $L\ge 2$. A length-$L$ LFSR can be written as a linear time-invariant system over $\F_2$:
\begin{align}
x_{t+1} &= A x_t,\qquad x_t\in \F_2^L,\; A\in \F_2^{L\times L}, \label{eq:lfsr_state}\\
y_t &= c^\top x_t,\qquad c\in \F_2^L\setminus\{0\}. \label{eq:lfsr_output}
\end{align}
When $A$ is the companion matrix of a primitive polynomial of degree $L$ and $x_0\neq 0$,
the output $\{y_t\}$ is an m-sequence (period $2^L-1$).

\subsection{Gold code (XOR of two m-sequences)}
Let two length-$L$ LFSRs be
\begin{align}
x^{(1)}_{t+1} &= A_1 x^{(1)}_t,\qquad a_t = c_1^\top x^{(1)}_t, \label{eq:lfsr1}\\
x^{(2)}_{t+1} &= A_2 x^{(2)}_t,\qquad b_t = c_2^\top x^{(2)}_t. \label{eq:lfsr2}
\end{align}
Define the Gold code output sequence
\begin{equation}
g_t = a_t \oplus b_t. \label{eq:gold}
\end{equation}
(Noise can be appended afterwards; the structural results below are noiseless.)

\subsection{History blocks}
For any binary sequence $\{u_t\}$ and integer $m\ge 1$, define the length-$m$ history block
\[
u_{t-m+1}^t := (u_{t-m+1},u_{t-m+2},\dots,u_t)\in \F_2^m.
\]

\section{Observability}

\begin{defn}[Observability matrix]
For the pair $(A,c^\top)$, define the length-$m$ observability matrix
\begin{equation}
\mathcal{O}_m(A,c^\top)
=
\begin{bmatrix}
c^\top\\
c^\top A\\
\vdots\\
c^\top A^{m-1}
\end{bmatrix}
\in \F_2^{m\times L}.
\end{equation}
The pair $(A,c^\top)$ is \emph{observable} if $\rank(\mathcal{O}_L(A,c^\top))=L$.
\end{defn}

\section{m-sequences: $L$ past bits determine the next bit}

\begin{thm}[Deterministic next-bit rule for an observable LFSR]
\label{thm:mseq_deterministic}
Assume $(A,c^\top)$ is observable. Then there exists a \emph{linear} function
$\ell:\F_2^L\to \F_2$ such that for all $t$,
\begin{equation}
y_{t+1} = \ell\big(y_{t-L+1}^t\big).
\end{equation}
Equivalently, the next bit is a deterministic function of the last $L$ bits.
\end{thm}

\begin{proof}
Consider the block of $L$ consecutive outputs starting at time $t-L+1$:
\begin{equation}
\begin{bmatrix}
y_{t-L+1}\\
y_{t-L+2}\\
\vdots\\
y_{t}
\end{bmatrix}
=
\begin{bmatrix}
c^\top\\
c^\top A\\
\vdots\\
c^\top A^{L-1}
\end{bmatrix}
x_{t-L+1}
= \mathcal{O}_L(A,c^\top)\, x_{t-L+1}. \label{eq:block_obs}
\end{equation}
Since $(A,c^\top)$ is observable, $\mathcal{O}_L(A,c^\top)$ is invertible over $\F_2$; hence
\begin{equation}
x_{t-L+1} = \mathcal{O}_L(A,c^\top)^{-1}
\begin{bmatrix}
y_{t-L+1}\\
\vdots\\
y_t
\end{bmatrix}. \label{eq:state_from_outputs}
\end{equation}
Then
\begin{align}
y_{t+1}
&= c^\top x_{t+1}
= c^\top A^{L} x_{t-L+1} \notag\\
&= c^\top A^{L} \mathcal{O}_L(A,c^\top)^{-1}
\begin{bmatrix}
y_{t-L+1}\\
\vdots\\
y_t
\end{bmatrix}. \label{eq:linear_predictor}
\end{align}
The right-hand side is a linear function of the last $L$ outputs, proving the claim.
\end{proof}

\section{Gold codes: a short history does not determine the next bit}

The key difference is that a Gold generator has a \emph{joint} state dimension $2L$.
Define the joint state and output map:
\begin{equation}
s_t := \begin{bmatrix} x^{(1)}_t\\ x^{(2)}_t\end{bmatrix}\in \F_2^{2L},\qquad
s_{t+1} = \begin{bmatrix} A_1&0\\0&A_2\end{bmatrix} s_t =: A s_t,
\qquad
g_t = \begin{bmatrix} c_1\\c_2\end{bmatrix}^\top s_t =: h^\top s_t.
\end{equation}

\subsection{A linear-algebra lemma for truncated observability}

\begin{lem}[Existence of indistinguishable states under short observation]
\label{lem:short_obs_kernel}
Let $(A,c^\top)$ be an observable pair of dimension $L$. Fix any integer $m<L$ and define
$\mathcal{O}_m=\mathcal{O}_m(A,c^\top)\in \F_2^{m\times L}$.
Then $\Ker(\mathcal{O}_m)$ is nontrivial, i.e.\ there exists $\Delta\in \F_2^L\setminus\{0\}$
such that
\begin{equation}
c^\top A^k \Delta = 0,\qquad k=0,1,\dots,m-1.
\end{equation}
\end{lem}

\begin{proof}
Since $\mathcal{O}_m$ has at most $m$ rows, $\rank(\mathcal{O}_m)\le m < L$.
By the rank--nullity theorem over $\F_2$,
\[
\dim \Ker(\mathcal{O}_m) = L-\rank(\mathcal{O}_m) \ge L-m \ge 1,
\]
so $\Ker(\mathcal{O}_m)$ contains a nonzero vector $\Delta$.
The condition $\mathcal{O}_m \Delta = 0$ is exactly $c^\top A^k \Delta=0$ for $k=0,\dots,m-1$.
\end{proof}

\subsection{Non-function property for Gold codes when the window is too short}

\begin{thm}[Gold code next bit is not determined by $m$ past bits for $m<L$]
\label{thm:gold_not_function_m_less_L}
Assume $(A_2,c_2^\top)$ is observable. Fix any $m<L$.
Then there exist two Gold sequences $\{g_t\}$ and $\{\tilde g_t\}$ (arising from different
initial states of the \emph{second} LFSR while keeping the first LFSR fixed) and a time index $t$
such that
\begin{equation}
g_{t-m+1}^t = \tilde g_{t-m+1}^t
\quad \text{but}\quad
g_{t+1}\neq \tilde g_{t+1}.
\end{equation}
Consequently, there does not exist any function $F:\F_2^m\to \F_2$ satisfying
$g_{t+1}=F(g_{t-m+1}^t)$ for all $t$ and all choices of seeds.
\end{thm}

\begin{proof}
Fix any realization of the first LFSR, hence fix $\{a_t\}$.
Consider two realizations of the second LFSR whose states at time $t-m+1$ differ by
$\Delta\in \F_2^L$:
\begin{equation}
\tilde x^{(2)}_{t-m+1} = x^{(2)}_{t-m+1}\oplus \Delta.
\end{equation}
By linearity of the state update,
\[
\tilde x^{(2)}_{t-m+1+k} = A_2^k \tilde x^{(2)}_{t-m+1}
= A_2^k x^{(2)}_{t-m+1}\oplus A_2^k\Delta,
\]
and thus the output difference satisfies
\begin{equation}
\tilde b_{t-m+1+k}\oplus b_{t-m+1+k}
= c_2^\top A_2^k \Delta,\qquad k\ge 0. \label{eq:diff_outputs}
\end{equation}
By Lemma~\ref{lem:short_obs_kernel}, since $m<L$ there exists $\Delta\neq 0$ such that
$c_2^\top A_2^k\Delta=0$ for $k=0,1,\dots,m-1$. With this choice,
\[
\tilde b_{\tau} = b_{\tau}\quad \text{for all }\tau\in\{t-m+1,\dots,t\}.
\]
Therefore, using $g_\tau=a_\tau\oplus b_\tau$ and $\tilde g_\tau=a_\tau\oplus \tilde b_\tau$,
\[
\tilde g_\tau = g_\tau\quad\text{for }\tau=t-m+1,\dots,t,
\]
i.e.\ $\tilde g_{t-m+1}^t = g_{t-m+1}^t$.

Because $\Delta\neq 0$ and $(A_2,c_2^\top)$ is observable, there exists some index $k\ge m$
such that $c_2^\top A_2^k\Delta=1$ (otherwise all outputs would be equal forever, contradicting
observability / distinguishability). In particular, there exists a smallest $k^\star\ge m$
with $c_2^\top A_2^{k^\star}\Delta=1$. Setting $t':=t-m+1+k^\star-1$ yields
\[
\tilde g_{t'-m+1}^{t'} = g_{t'-m+1}^{t'}
\quad\text{but}\quad
\tilde g_{t'+1}\neq g_{t'+1}.
\]
Hence no single-valued function of $m$ past bits can represent the next bit for all seeds.
\end{proof}

\section{Deterministic next-bit rule exists once $m$ reaches the joint state dimension}

\begin{thm}[Gold generator becomes predictable from a long enough history]
\label{thm:gold_predictable_when_observable}
Consider the joint system $(A,h^\top)$ of dimension $2L$:
\[
s_{t+1}=As_t,\qquad g_t=h^\top s_t.
\]
If $\rank(\mathcal{O}_m(A,h^\top)) = 2L$ for some $m$, then there exists a (linear) function
$F:\F_2^m\to\F_2$ such that
\[
g_{t+1} = F(g_{t-m+1}^t)\quad\text{for all }t.
\]
\end{thm}

\begin{proof}
Let $\mathcal{O}_m=\mathcal{O}_m(A,h^\top)\in \F_2^{m\times 2L}$.
The length-$m$ output block satisfies
\[
g_{t-m+1}^t = \mathcal{O}_m s_{t-m+1}.
\]
If $\rank(\mathcal{O}_m)=2L$, then $\mathcal{O}_m$ has full column rank, hence there exists
a left-inverse $\mathcal{O}_m^\dagger\in \F_2^{2L\times m}$ such that
$\mathcal{O}_m^\dagger \mathcal{O}_m = I_{2L}$.
Thus
\[
s_{t-m+1}=\mathcal{O}_m^\dagger g_{t-m+1}^t,
\]
and therefore
\[
g_{t+1} = h^\top s_{t+1}
= h^\top A^{m} s_{t-m+1}
= h^\top A^{m}\mathcal{O}_m^\dagger g_{t-m+1}^t,
\]
which is a linear function of the past $m$ outputs.
\end{proof}

\section{Remarks (interpretation)}
\begin{itemize}
\item Theorem~\ref{thm:mseq_deterministic} shows that for an m-sequence (state dimension $L$),
a context length $m=L$ suffices for deterministic prediction.
\item Theorem~\ref{thm:gold_not_function_m_less_L} shows that for a Gold code (joint state dimension $2L$),
any \emph{short} context $m<L$ cannot determine the next bit uniformly over all seeds.
\item Theorem~\ref{thm:gold_predictable_when_observable} shows that once the joint system is observable
from output history (typically for $m$ on the order of $2L$), a deterministic rule exists again.
\end{itemize}
